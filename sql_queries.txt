1. Створення бази даних.
CREATE SCHEMA LibraryManagement;
USE LibraryManagement;

CREATE TABLE authors (
author_id INT AUTO_INCREMENT PRIMARY KEY,
author_name VARCHAR(255)
);
CREATE TABLE genres (
genre_id INT AUTO_INCREMENT PRIMARY KEY,
genre_name VARCHAR(255)
);
CREATE TABLE books (
book_id INT AUTO_INCREMENT PRIMARY KEY,
title VARCHAR(255),
publication_year YEAR,
author_id INT,
genre_id INT,
FOREIGN KEY (author_id) REFERENCES authors(author_id),
FOREIGN KEY (genre_id) REFERENCES genres(genre_id)
);
CREATE TABLE users (
user_id INT AUTO_INCREMENT PRIMARY KEY,
username VARCHAR(255),
email VARCHAR(255)
);
CREATE TABLE borrowed_books (
borrow_id INT AUTO_INCREMENT PRIMARY KEY,
book_id INT,
user_id INT,
borrow_date DATE,
return_date DATE,
FOREIGN KEY (book_id) REFERENCES books(book_id),
FOREIGN KEY (user_id) REFERENCES users(user_id)
);

4.2. Змінення декілька операторів INNER на LEFT чи RIGHT.
USE hw_03;

SELECT COUNT(*) AS total_rows
FROM order_details
RIGHT JOIN orders ON order_details.order_id = orders.id
RIGHT JOIN customers ON orders.customer_id = customers.id
LEFT JOIN employees ON orders.employee_id = employees.employee_id
LEFT JOIN shippers ON orders.shipper_id = shippers.id
LEFT JOIN products ON order_details.product_id = products.id
LEFT JOIN categories ON products.category_id = categories.id
LEFT JOIN suppliers ON products.supplier_id = suppliers.id;

Результат: 535 рядків (замість 518). Кількість рядків збільшилась, тому що INNER JOIN повертає тільки ті рядки,
які мають відповідність в обох таблицях, а LEFT/RIGHT JOIN повертають усі рядки
з лівої/правої таблиці, навіть якщо вони не мають відповідності в іншій таблиці. Якщо немає відповідності в іншій таблиці, виводяться значення NULL.

4.3. Вибір рядків, де employee_id > 3 та ≤ 10.
USE hw_03;

SELECT COUNT(*) AS total_rows
FROM order_details
INNER JOIN orders ON order_details.order_id = orders.id
INNER JOIN customers ON orders.customer_id = customers.id
INNER JOIN employees ON orders.employee_id = employees.employee_id
INNER JOIN shippers ON orders.shipper_id = shippers.id
INNER JOIN products ON order_details.product_id = products.id
INNER JOIN categories ON products.category_id = categories.id
INNER JOIN suppliers ON products.supplier_id = suppliers.id
WHERE orders.employee_id > 3 AND orders.employee_id <= 10;

4.4. Групування за іменем категорії, визначення кількості рядків у групі та середньої кількості товару.
USE hw_03;

SELECT categories.name AS category_name, COUNT(*) AS total_rows, AVG(order_details.quantity) AS average_quantity
FROM order_details
INNER JOIN products ON order_details.product_id = products.id
INNER JOIN categories ON products.category_id = categories.id
GROUP BY categories.name;

4.5. Фільтрація рядків, де середня кількість товару більша за 21.
USE hw_03;

SELECT categories.name AS category_name, COUNT(*) AS total_rows, AVG(order_details.quantity) AS average_quantity
FROM order_details
INNER JOIN products ON order_details.product_id = products.id
INNER JOIN categories ON products.category_id = categories.id
GROUP BY categories.name
HAVING average_quantity > 21;

4.6. Сортування за спаданням кількості рядків.
